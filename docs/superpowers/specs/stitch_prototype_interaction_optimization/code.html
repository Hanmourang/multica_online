-- 逻辑表不落数据，定义即虚拟视图；运行时直接下推计算 CREATE OR REPLACE VIEW lg_issue_detail_view AS SELECT i.workspace_id, i.id AS issue_id, i.priority_score, AVG(r.progress_pct) AS avg_progress_pct, COUNT(a.id) AS deliverable_cnt, MAX(o.synced_at) AS last_synced_at FROM issue i LEFT JOIN agent_run r ON r.issue_id = i.id LEFT JOIN project_artifact a ON a.issue_id = i.id LEFT JOIN ods_issue_sync_raw o ON o.issue_id = i.id GROUP BY i.workspace_id, i.id;
